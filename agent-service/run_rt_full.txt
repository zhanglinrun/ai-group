   102	async def _run_graph_with_progress_heartbeat(
   103	    *,
   104	    run_id: str,
   105	    phase: str,
   106	    graph: Any,
   107	    config: dict[str, object],
   108	    invoke_coro: Awaitable[Any],
   109	) -> Any:
   110	    """Emit structlog heartbeats while a long graph.ainvoke is in flight."""
   111	    started_at = datetime.now(timezone.utc)
   112	    stop_event = asyncio.Event()
   113	
   114	    async def _heartbeat_loop() -> None:
   115	        while not stop_event.is_set():
   116	            try:
   117	                await asyncio.wait_for(stop_event.wait(), timeout=_RUN_PROGRESS_INTERVAL_SECONDS)
   118	            except asyncio.TimeoutError:
   119	                checkpoint_next: str | None = None
   120	                try:
   121	                    snapshot = await graph.aget_state(config)
   122	                    if snapshot.next:
   123	                        checkpoint_next = str(snapshot.next[0])
   124	                except (AttributeError, RuntimeError, TypeError, ValueError):
   125	                    checkpoint_next = None
   126	                log.debug(
   127	                    "run.progress",
   128	                    run_id=run_id,
   129	                    phase=phase,
   130	                    elapsed_ms=int(
   131	                        (datetime.now(timezone.utc) - started_at).total_seconds() * 1000
   132	                    ),
   133	                    checkpoint_next=checkpoint_next,
   134	                )
   135	
   136	    heartbeat_task = asyncio.create_task(_heartbeat_loop(), name=f"run_progress_{run_id}")
   137	    try:
   138	        return await invoke_coro
   139	    finally:
   140	        stop_event.set()
   141	        heartbeat_task.cancel()
   142	        with contextlib.suppress(asyncio.CancelledError):
   143	            await heartbeat_task

   803	async def _mark_run_paused_for_quota(*, run_id: str, log_event: str) -> None:
   804	    """Stop the graph without failing it so the user can recharge and resume."""
   805	    log.info(log_event, error_type="QuotaExhaustedError", status="paused")
   806	    session_factory = get_session_factory()
   807	    async with session_factory() as session:
   808	        run = await session.get(Run, run_id)
   809	        if run is None:
   810	            return
   811	        if run.status in {"completed", "degraded", "failed", "cancelled"}:
   812	            return

   818	        run_id=run_id,
   819	        event_type=RunEventType.RUN_FINISH,
   820	        payload=_build_run_finish_payload(
   821	            run_id=run_id,

   823	            error_type="QuotaExhaustedError",
   824	            error_message=QUOTA_PAUSED_REASON,

   837	async def _mark_run_failed_and_emit(
   838	    *,
   839	    run_id: str,
   840	    exc: BaseException,
   841	    log_event: str,
   842	) -> None:
   843	    """Background-task boundary cleanup: persist run.status=failed + emit RUN_FINISH.
   844	
   845	    Centralises the failure path so all three async graph runners stay in lockstep
   846	    when a node throws an unexpected error. Without this the asyncio task dies
   847	    silently ("Task exception was never retrieved") and the Run row stays
   848	    "running" forever, leaving the UI polling against a corpse.
   849	    """
   850	    error_type = type(exc).__name__
   851	    error_message = format_exception_for_log(exc)
   852	    status_reason = humanize_failure_message(error_message)
   853	    log.error(log_event, error_type=error_type, error=error_message)
   854	    session_factory = get_session_factory()
   855	    async with session_factory() as session:
   856	        run = await session.get(Run, run_id)
   857	        if run is not None:
   858	            run.status = "failed"
   859	            run.status_reason = status_reason
   860	            run.finished_at = datetime.now(timezone.utc)
   861	            await session.commit()
   862	    await settle_run_billing(run_id=run_id, terminal_status="failed")
   863	    await emit_run_event(
   864	        run_id=run_id,
   865	        event_type=RunEventType.RUN_FINISH,
   866	        payload=_build_run_finish_payload(
   867	            run_id=run_id,
   868	            status="failed",
   869	            error_type=error_type,
   870	            error_message=status_reason,
   871	            status_reason=status_reason,
   872	        ),
   873	    )

   883	async def _handle_graph_cancelled(*, run_id: str, log_event: str) -> None:
   884	    """Reconcile the Run row when a graph task receives CancelledError.
   885	
   886	    Two ways this fires:
   887	      1. User cancel via PATCH /runs — the endpoint already flipped the row to
   888	         "cancelled" and emitted RUN_FINISH, so this branch is a no-op.
   889	      2. Unexpected cancellation (e.g. lifespan shutdown asking tasks to stop) —
   890	         the row is still "running"; mark it failed so the UI doesn't hang.
   891	    The caller MUST re-raise CancelledError per asyncio's cooperative-cancel
   892	    contract; failing to do so makes the task look like a normal completion.
   893	    """
   894	    session_factory = get_session_factory()
   895	    async with session_factory() as session:
   896	        run = await session.get(Run, run_id)
   897	        if run is None:
   898	            return
   899	        if run.status != "running":
   900	            with bind_run(run_id):
   901	                log.info(log_event, status=run.status, branch="already_terminal")
   902	            if run.status not in {"paused"}:
   903	                await settle_run_billing(run_id=run_id, terminal_status=str(run.status))
   904	            return
   905	        run.status = "failed"
   906	        run.status_reason = UNEXPECTED_CANCEL_REASON
   907	        run.finished_at = datetime.now(timezone.utc)
   908	        await session.commit()
   909	    await settle_run_billing(run_id=run_id, terminal_status="failed")
   910	    with bind_run(run_id):
   911	        log.warning(log_event, status="failed", branch="unexpected_cancel")
   912	    await emit_run_event(
   913	        run_id=run_id,
   914	        event_type=RunEventType.RUN_FINISH,
   915	        payload=_build_run_finish_payload(
   916	            run_id=run_id,
   917	            status="failed",
   918	            error_type="CancelledError",
   919	            error_message=UNEXPECTED_CANCEL_REASON,
   920	            status_reason=UNEXPECTED_CANCEL_REASON,
   921	        ),
   922	    )

   925	def _cancel_background_tasks_for_run(
   926	    *,
   927	    background_tasks: set[asyncio.Task[object]] | None,
   928	    run_id: str,
   929	) -> int:
   930	    """Cancel any in-flight graph tasks bound to this run.
   931	
   932	    Tasks are named with the run_id suffix on creation (see `name=f"..._{run_id}"`
   933	    in this module). We don't cancel the skill_curator follow-up — it only fires
   934	    on terminal completion, so a cancel during execution means there's no curator
   935	    task to chase yet. CancelledError propagates back through the outer boundary
   936	    (`_mark_run_failed_and_emit`), which would normally flip the row to failed;
   937	    PATCH /runs caller flips it to "cancelled" first so the user's intent wins.
   938	    """
   939	    if not isinstance(background_tasks, set):
   940	        return 0
   941	    cancelled = 0
   942	    for task in list(background_tasks):
   943	        name = task.get_name()
   944	        if not any(name.startswith(prefix) and name.endswith(run_id) for prefix in _RUN_TASK_NAME_PREFIXES):
   945	            continue
   946	        if task.done():
   947	            continue
   948	        task.cancel()
   949	        cancelled += 1
   950	    return cancelled

  1033	async def _run_event_stream(
  1034	    *,
  1035	    event_bus: EventBus,
  1036	    run_id: str,
  1037	    keepalive_seconds: float = 15.0,
  1038	    max_events: int | None = None,
  1039	) -> AsyncIterator[str]:
  1040	    yield "retry: 15000\n\n"
  1041	    emitted_count = 0
  1042	    async with event_bus.subscribe(run_id) as queue:
  1043	        while True:
  1044	            try:
  1045	                event = await asyncio.wait_for(queue.get(), timeout=keepalive_seconds)
  1046	            except asyncio.TimeoutError:
  1047	                yield ": keepalive\n\n"
  1048	                continue
  1049	            except asyncio.CancelledError:
  1050	                return
  1051	            yield _to_sse_chunk(
  1052	                event=event.event_type.value,
  1053	                data=event.model_dump(mode="json"),
  1054	            )
  1055	            emitted_count += 1
  1056	            if max_events is not None and emitted_count >= max_events:
  1057	                return

  1066	def _derive_run_phase(run: Run) -> Literal["intake", "planning", "executing", "done"] | None:
  1067	    """Phase is a derived view, not stored — the source of truth is the LangGraph state.
  1068	
  1069	    Legacy runs (created via POST /api/runs without intake) have `intake_draft is None`
  1070	    and return `None` here so the FE renders them with the old layout.
  1071	
  1072	    Phase 2: `plan_tree.confirmed_at` is the planning→executing signal. planner_generate
  1073	    writes `confirmed_at=None`; planner_wait sets it on resume.
  1074	    """
  1075	    intake_draft = run.intake_draft
  1076	    if intake_draft is None:
  1077	        return None
  1078	    if run.status in {"completed", "degraded", "failed", "cancelled"}:
  1079	        return "done"
  1080	    plan_tree = run.plan_tree
  1081	    if plan_tree is not None:
  1082	        if plan_tree.get("confirmed_at") is not None:
  1083	            return "executing"
  1084	        return "planning"
  1085	    intake_complete = bool(intake_draft.get("user_role")) and bool(
  1086	        intake_draft.get("analysis_intent")
  1087	    ) and (
  1088	        bool(intake_draft.get("competitors_explicit"))
  1089	        or bool(intake_draft.get("competitors_discovery_mode"))
  1090	    )
  1091	    if not intake_complete:
  1092	        return "intake"
  1093	    return "planning"

  1325	def _normalize_competitor_key(value: str) -> str:
  1326	    lowered = value.casefold().strip()
  1327	    if not lowered:
  1328	        return ""
  1329	    normalized = re.sub(r"[^\w\s\u4e00-\u9fff]+", " ", lowered, flags=re.UNICODE)
  1330	    tokens = [token for token in normalized.split() if token]
  1331	    if not tokens:
  1332	        return ""
  1333	    # Token sorting absorbs punctuation and token-order variants
  1334	    # ("Meta Ray-Ban" vs "Ray-Ban Meta").
  1335	    return " ".join(sorted(tokens))

  1890	def _normalize_idempotency_key(
  1891	    *,
  1892	    header_value: str | None,
  1893	    body_value: str | None,
  1894	) -> str:
  1895	    """Resolve create-request idempotency key with strict precedence.
  1896	
  1897	    Header key is preferred so upstream gateways / SDK retries can inject it
  1898	    without mutating JSON payloads. Falls back to body field for clients that
  1899	    cannot set custom headers.
  1900	    """
  1901	    header_key = header_value.strip() if isinstance(header_value, str) else ""
  1902	    if header_key:
  1903	        return header_key
  1904	    body_key = body_value.strip() if isinstance(body_value, str) else ""
  1905	    if body_key:
  1906	        return body_key
  1907	    return f"idemp_{uuid4().hex}"
